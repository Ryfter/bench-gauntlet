def line_kg(s):
    if s is None:
        return None
    if type(s) is not str:
        return None
    t = s.strip().lower()
    if t == "" or t == "n/a":
        return 0.0
    t2 = ""
    for ch in t:
        if ch != " ":
            t2 = t2 + ch
    t = t2
    if len(t) >= 2 and t[-2:] == "kg":
        num = t[:-2]
        if num == "" or num == "-" or num == "+" or num == ".":
            return None
        try:
            v = float(num)
        except ValueError:
            return None
        if v < 0:
            return None
        return float(v)
    if len(t) >= 1 and t[-1] == "g":
        num = t[:-1]
        if num == "" or num == "-" or num == "+" or num == ".":
            return None
        try:
            v = float(num)
        except ValueError:
            return None
        if v < 0:
            return None
        return float(v) / 1000.0
    return None
