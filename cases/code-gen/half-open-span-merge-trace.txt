def merge_spans(spans):
    if not spans:
        return []
    items = sorted(spans, key=lambda p: (p[0], p[1]))
    out = [list(items[0])]
    for s, e in items[1:]:
        if s <= out[-1][1]:
            if e > out[-1][1]:
                out[-1][1] = e
        else:
            out.append([s, e])
    return out
