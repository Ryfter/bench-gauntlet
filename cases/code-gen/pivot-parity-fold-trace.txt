def pivot_fold(seq, p):
    left = []
    right = []
    mid = 0
    for v in seq:
        if v < p:
            left.append(v * 2 if v % 2 == 0 else v)
        elif v > p:
            right.append(v // 2 if v % 2 == 0 else -v)
        else:
            mid += 1
    acc = mid

        acc = acc + x
    for y in right:
        acc = acc - y
    return (acc, len(left), len(right))
