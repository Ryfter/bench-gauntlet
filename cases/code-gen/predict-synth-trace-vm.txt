def synth_trace(cmds):
    reg = 0
    i = 0
    while i < len(cmds):
        parts = cmds[i].split()
        op = parts[0]
        if op == "SET":
            reg = int(parts[1])
            i += 1
        elif op == "ADD":
            reg = reg + int(parts[1])
            i += 1
        elif op == "MUL":
            reg = reg * int(parts[1])
            i += 1
        elif op == "DIV":
            x = int(parts[1])
            if x != 0:
                reg = reg // x
            i += 1
        elif op == "NEG":
            reg = -reg
            i += 1
        elif op == "JMP":
            n = int(parts[1])
            if reg > 0:
                i += 1 + n
            else:
                i += 1
        else:
            i += 1
    return reg
