def free_blocks(n, taken):
    occ = set()
    for i in taken:
        if isinstance(i, int) and 0 <= i < n:
            occ.add(i)
            if i + 1 < n:
                occ.add(i + 1)
    out = []
    i = 0
    while i < n:
        if i in occ:
            i += 1
            continue
        j = i
        while j < n and j not in occ:
            j += 1
        out.append((i, j - i))
        i = j
    return out
